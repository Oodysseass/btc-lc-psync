%%% Theorem environments %%%
%% llncs predefines theorem, lemma, corollary, definition, remark and proof,
%% each numbered with its own counter. We add assumption in the same style
%% and register its name with cleveref. The proof environment of llncs prints
%% no end mark, so we append one.
\spnewtheorem{assumption}{Assumption}{\bfseries}{\itshape}
\crefname{assumption}{assumption}{assumptions}
\Crefname{assumption}{Assumption}{Assumptions}
\let\llncsendproof\endproof
\renewcommand{\endproof}{\qed\llncsendproof}
